cmake_minimum_required(VERSION 3.18)
project(KoettMeetingWhisper LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${WHISPER_SOURCE}" whisper EXCLUDE_FROM_ALL)
add_library(KoettMeetingWhisper MODULE bridge.c)
set_target_properties(KoettMeetingWhisper PROPERTIES SUFFIX ".dylib" MACOSX_BUNDLE FALSE)
target_link_libraries(KoettMeetingWhisper PRIVATE whisper)
